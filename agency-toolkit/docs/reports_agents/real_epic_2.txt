```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any
from pathlib import Path


@dataclass
class TaskContext:
    """Execution context for a task.

    Attributes:
        config: Global Config object
        step_context: Outputs from previous tasks (e.g., {"tagline": "..."})
        archetype_id: Current archetype being processed (optional)
        solution_id: Current solution being executed (optional)
    """
    config: Any  # TODO: Import Config type from models
    step_context: dict[str, Any]
    archetype_id: str | None = None
    solution_id: str | None = None


class TaskHandler(ABC):
    """Base class for all task handlers (Strategy Pattern).

    Example:
        class MyTaskHandler(TaskHandler):
            @property
            def name(self) -> str:
                return "mytask"

            def validate_params(self, params: dict) -> None:
                if "required_field" not in params:
                    raise ValueError("Missing required_field")

            def execute(self, params: dict, context: TaskContext) -> dict:
                return {"result": "success"}
    """

    @property
    @abstractmethod
    def name(self) -> str:
        """Unique task name (e.g., 'ai', 'social')."""
        pass

    @abstractmethod
    def validate_params(self, params: dict[str, Any]) -> None:
        """Validate params before execution.

        Args:
            params: Task parameters from JSON

        Raises:
            ValueError: If params are invalid
        """
        pass

    @abstractmethod
    def execute(
        self,
        params: dict[str, Any],
        context: TaskContext
    ) -> dict[str, Any]:
        """Execute the task.

        Args:
            params: Task parameters (already validated)
            context: Execution context with config and step outputs

        Returns:
            Task output (will be stored in step_context if output_key set)
        """
        pass

RULES:
- Keep it simple (no fancy logic yet)
- ABC forces subclasses to implement all methods
- TaskContext is dataclass (not Pydantic) for speed
```

```python
from typing import Type
from .base import TaskHandler


# Global registry (same pattern as providers/provider_loader.py)
_TASK_HANDLERS: dict[str, Type[TaskHandler]] = {}


def register_task_handler(handler_class: Type[TaskHandler]) -> None:
    """Register a task handler class.

    Args:
        handler_class: TaskHandler subclass

    Example:
        register_task_handler(AITaskHandler)
    """
    handler_instance = handler_class()
    name = handler_instance.name
    _TASK_HANDLERS[name] = handler_class


def get_task_handler(name: str) -> TaskHandler:
    """Get a task handler instance by name.

    Args:
        name: Task name (e.g., 'ai', 'social')

    Returns:
        Task handler instance

    Raises:
        ValueError: If handler not found

    Example:
        handler = get_task_handler("ai")
        result = handler.execute(params, context)
    """
    if name not in _TASK_HANDLERS:
        available = ", ".join(_TASK_HANDLERS.keys())
        raise ValueError(
            f"Unknown task handler: '{name}'. "
            f"Available handlers: {available}"
        )

    handler_class = _TASK_HANDLERS[name]
    return handler_class()


def list_task_handlers() -> list[str]:
    """List all registered task handler names."""
    return sorted(_TASK_HANDLERS.keys())

RULES:
- Same pattern as providers/provider_loader.py
- Clear error messages (show available handlers)
- list_task_handlers() for CLI help

TEST:
- Register dummy handler
- Retrieve it
- Verify ValueError for unknown handler
```

```python
def _execute_task(tool_name: str, task: dict, context: dict) -> Any:
    """Execute a task using the handler registry.

    Args:
        tool_name: Task handler name (e.g., 'ai')
        task: Task definition with params
        context: Execution context (will be converted to TaskContext)

    Returns:
        Task output
    """
    # Get handler (raises ValueError if unknown)
    handler = get_task_handler(tool_name)

    # Extract params and create typed context
    params = task.get("params", {})
    task_context = TaskContext(
        config=context.get("config"),
        step_context=context.get("step_context", {}),
        archetype_id=context.get("archetype_id"),
        solution_id=context.get("solution_id")
    )

    # Validate and execute
    handler.validate_params(params)
    output = handler.execute(params, task_context)

    return output
```
