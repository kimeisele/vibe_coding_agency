## General Principles

### SOLID Principles
- **Single Responsibility**: Each function/class does ONE thing
- **Open/Closed**: Open for extension, closed for modification
- **Liskov Substitution**: Subtypes must be substitutable for base types
- **Interface Segregation**: Many specific interfaces > one general interface
- **Dependency Inversion**: Depend on abstractions, not concretions

### DRY (Don't Repeat Yourself)
- Extract repeated code into functions
- Use configuration for repeated values
- Create base classes for shared behavior

### YAGNI (You Aren't Gonna Need It)
- Implement only what's required now
- Don't add "future" features
- Refactor when needs actually arise

### Type Hints

ALWAYS use type hints:

```python
def create_user(name: str, age: int) -> User:
    """Create a new user.
    
    Args:
        name: User's full name
        age: User's age in years
        
    Returns:
        Created User instance
        
    Raises:
        ValidationError: If name is empty or age is negative
    """
    if not name:
        raise ValidationError("Name cannot be empty")
    if age < 0:
        raise ValidationError("Age cannot be negative")
    
    return User(name=name, age=age)
```

### Docstrings

Use Google-style docstrings:

```python
def complex_function(param1: str, param2: int) -> dict:

            'key1': 'value type',

        }

        ValueError: When param1 is empty
        TypeError: When param2 is negative
        
    Example:
        >>> result = complex_function("test", 42)

    pass
```

### Error Handling

**DO**:

```python
try:
    result = risky_operation()
except SpecificException as e:
    logger.error(f"Operation failed: {e}")
    raise CustomException("Helpful message") from e
```

**DON’T**:

```python
try:
    result = risky_operation()
except:  # Too broad!
    pass  # Silent failure!
```

### Logging

        logger.error(f"Processing failed: {e}", exc_info=True)

## Architecture Patterns

### Repository Pattern

```python
# repositories/base_repository.py
from abc import ABC, abstractmethod
from typing import Generic, TypeVar, List, Optional

T = TypeVar('T')

class BaseRepository(ABC, Generic[T]):
    """Base repository interface."""
    
    @abstractmethod
    def get(self, id: str) -> Optional[T]:
        """Get entity by ID."""
        pass
    
    @abstractmethod
    def list(self, limit: int = 100) -> List[T]:
        """List entities."""
        pass
    
    @abstractmethod
    def create(self, entity: T) -> T:
        """Create entity."""
        pass
    
    @abstractmethod
    def update(self, id: str, entity: T) -> T:
        """Update entity."""
        pass
    
    @abstractmethod
    def delete(self, id: str) -> None:
        """Delete entity."""
        pass

# repositories/user_repository.py
from src.models.user import User
from src.repositories.base_repository import BaseRepository

class UserRepository(BaseRepository[User]):
    """User data access."""
    
    def __init__(self, session):
        self.session = session
    
    def get(self, id: str) -> Optional[User]:
        return self.session.query(User).filter(User.id == id).first()
    
    # ... implement other methods
```

### Service Layer

```python
# services/user_service.py
from typing import Optional
from src.models.user import User
from src.repositories.user_repository import UserRepository
from src.schemas.user import UserCreate

class UserService:
    """User business logic."""
    
    def __init__(self, user_repository: UserRepository):
        self.user_repository = user_repository
    
    def create_user(self, user_data: UserCreate) -> User:
        """Create a new user.
        
        Args:
            user_data: User creation data
            
        Returns:
            Created user
            
        Raises:
            ValidationError: If email already exists
        """
        # Business rule: email must be unique
        existing = self.user_repository.get_by_email(user_data.email)
        if existing:
            raise ValidationError("Email already in use")
        
        user = User(**user_data.dict())
        return self.user_repository.create(user)
```

### Dependency Injection

```python
# api/dependencies.py
from functools import lru_cache
from sqlalchemy.orm import Session

from src.database import SessionLocal
from src.repositories.user_repository import UserRepository
from src.services.user_service import UserService

def get_db() -> Session:
    """Get database session."""
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@lru_cache()
def get_user_service(db: Session = Depends(get_db)) -> UserService:
    """Get user service with dependencies."""
    user_repo = UserRepository(db)
    return UserService(user_repo)

# api/routes/users.py
from fastapi import APIRouter, Depends

router = APIRouter()

@router.post("/users")
def create_user(
    user_data: UserCreate,
    service: UserService = Depends(get_user_service)
):
    """Create user endpoint."""
    return service.create_user(user_data)
```

import pytest
from unittest.mock import Mock

from src.services.user_service import UserService
from src.schemas.user import UserCreate

class TestUserService:
    """Test UserService business logic."""
    
    @pytest.fixture
    def mock_repository(self):
        """Create mock repository."""
        return Mock()
    
    @pytest.fixture
    def service(self, mock_repository):
        """Create service with mocked dependencies."""
        return UserService(mock_repository)
    
    def test_create_user_success(self, service, mock_repository):
        """Test successful user creation."""
        # Arrange
        user_data = UserCreate(name="Test", email="test@example.com")
        mock_repository.get_by_email.return_value = None
        
        # Act
        result = service.create_user(user_data)
        
        # Assert
        assert result.name == "Test"
        mock_repository.create.assert_called_once()
    
    def test_create_user_duplicate_email(self, service, mock_repository):
        """Test user creation with duplicate email."""
        # Arrange
        user_data = UserCreate(name="Test", email="test@example.com")
        mock_repository.get_by_email.return_value = Mock()  # Email exists
        
        # Act & Assert
        with pytest.raises(ValidationError, match="Email already in use"):
            service.create_user(user_data)

## Security Standards

### Input Validation

```python
from pydantic import BaseModel, Field, validator

class UserCreate(BaseModel):
    """User creation schema with validation."""
    
    name: str = Field(..., min_length=1, max_length=100)
    email: str = Field(..., regex=r'^[\w\.-]+@[\w\.-]+\.\w+$')
    age: int = Field(..., ge=0, le=150)
    
    @validator('email')
    def email_must_be_lowercase(cls, v):
        """Ensure email is lowercase."""
        return v.lower()
```
